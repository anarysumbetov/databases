-- 1
CREATE OR REPLACE VIEW flights_by_date AS
SELECT
    f.flight_id,
    a.airline_name,
    dep.airport_name AS departing_airport,
    arr.airport_name AS arriving_airport,
    f.sch_departure_time,
    f.sch_arrival_time,
    f.departing_gate,
    f.arriving_gate
FROM flights f
JOIN airline a
    ON f.airline_id = a.airline_id
JOIN airport dep
    ON f.departing_airport_id = dep.airport_id
JOIN airport arr
    ON f.arriving_airport_id = arr.airport_id;

SELECT *
FROM flights_by_date
WHERE sch_departure_time::date = '2026-09-23';
-- 2
CREATE OR REPLACE VIEW bookings_next_week AS
SELECT
    b.booking_id,
    b.passenger_id,
    b.booking_platform,
    b.status AS booking_status,
    b.price,
    f.flight_id,
    a.airline_name,
    dep.airport_name AS departing_airport,
    arr.airport_name AS arriving_airport,
    f.sch_departure_time,
    f.sch_arrival_time
FROM booking b
JOIN flights f
    ON b.flight_id = f.flight_id
JOIN airline a
    ON f.airline_id = a.airline_id
JOIN airport dep
    ON f.departing_airport_id = dep.airport_id
JOIN airport arr
    ON f.arriving_airport_id = arr.airport_id
WHERE f.sch_departure_time >= CURRENT_DATE
    AND f.sch_departure_time < CURRENT_DATE + INTERVAL '7 days';

SELECT * FROM bookings_next_week;

-- 3
CREATE OR REPLACE VIEW top_5_popular_routes AS
SELECT
    dep.airport_name AS departing_airport,
    arr.airport_name AS arriving_airport,
    COUNT(b.booking_id) AS total_bookings
FROM flights f
JOIN airport dep
    ON f.departing_airport_id = dep.airport_id
JOIN airport arr
    ON f.arriving_airport_id = arr.airport_id
JOIN booking b
    ON f.flight_id = b.flight_id
GROUP BY
    dep.airport_name,
    arr.airport_name
ORDER BY total_bookings DESC
LIMIT 5;

SELECT * FROM top_5_popular_routes;

-- 6
CREATE OR REPLACE VIEW delayed_flights_over_24_hours AS
SELECT
    f.flight_id,
    a.airline_name,
    dep.airport_name AS departing_airport,
    arr.airport_name AS arriving_airport,
    f.sch_departure_time,
    f.act_departure_time,
    f.act_departure_time - f.sch_departure_time AS delay_duration
FROM flights f
JOIN airline a
    ON f.airline_id = a.airline_id
JOIN airport dep
    ON f.departing_airport_id = dep.airport_id
JOIN airport arr
    ON f.arriving_airport_id = arr.airport_id
WHERE f.act_departure_time IS NOT NULL
    AND f.act_departure_time > f.sch_departure_time + INTERVAL '24 hours';

SELECT *
FROM delayed_flights_over_24_hours;

-- 7
CREATE OR REPLACE VIEW leffler_thompson_passengers AS
SELECT DISTINCT
    p.passenger_id,
    p.first_name || ' ' || p.last_name AS full_name,
    p.country_of_citizenship AS country_of_origin
FROM passengers p
JOIN booking b
    ON p.passenger_id = b.passenger_id
WHERE b.booking_platform = 'Leffler-Thompson';

SELECT *
FROM leffler_thompson_passengers;

-- 8
CREATE OR REPLACE VIEW top_10_visited_countries AS
SELECT
    arr.country,
    COUNT(b.booking_id) AS total_visits
FROM booking b
JOIN flights f
    ON b.flight_id = f.flight_id
JOIN airport arr
    ON f.arriving_airport_id = arr.airport_id
GROUP BY arr.country
ORDER BY total_visits DESC
LIMIT 10;

SELECT *
FROM top_10_visited_countries;

-- 9
CREATE OR REPLACE VIEW delayed_flights_over_24_hours AS
SELECT
    f.flight_id,
    a.airline_name,
    dep.airport_name AS departing_airport,
    arr.airport_name AS arriving_airport,
    f.sch_departure_time,
    f.act_departure_time,
    f.act_departure_time - f.sch_departure_time AS delay_duration,
    EXTRACT(EPOCH FROM (f.act_departure_time - f.sch_departure_time)) / 3600 AS delay_hours
FROM flights f
JOIN airline a
    ON f.airline_id = a.airline_id
JOIN airport dep
    ON f.departing_airport_id = dep.airport_id
JOIN airport arr
    ON f.arriving_airport_id = arr.airport_id
WHERE f.act_departure_time IS NOT NULL
    AND f.act_departure_time > f.sch_departure_time + INTERVAL '24 hours';

SELECT *
FROM delayed_flights_over_24_hours;
