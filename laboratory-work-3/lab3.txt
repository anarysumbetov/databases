INSERT INTO airport (
	airport_id,
	airport_name,
	country,
	state,
	city,
	created_at,
	updated_at
)
SELECT 
	m.max_id + gs,
	'Airport_' || gs,
	'Country_' || gs,
	'State_' || gs,
	'City_' || gs,
	CURRENT_TIMESTAMP,
	CURRENT_TIMESTAMP
FROM generate_series(1, 200) AS gs
CROSS JOIN (SELECT COALESCE(MAX(airport_id), 0) AS max_id FROM airport) m;

SELECT COUNT(*) FROM airport;

FROM (SELECT COALESCE(MAX(airline_id), 0) AS max_id FROM airline) m

	m.max_id + 3,
	'FF',
	'FlyFly',
	'Poland',
	CURRENT_TIMESTAMP,
	CURRENT_TIMESTAMP
FROM (SELECT COALESCE(MAX(airline_id), 0) AS max_id FROM airline) m;

FROM airline
WHERE airline_name IN ('AirEasy', 'FLyHigh', 'FlyFly');
