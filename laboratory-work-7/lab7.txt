CREATE INDEX idx_booking_flight
ON booking_flight (booking_id, flight_id);

SELECT 
	f.flight_id,
	b.price
FROM booking b
JOIN booking_flight bf
	ON b.booking_id = bf.booking_id
JOIN flights f
	ON bf.flight_id = f.flight_id
WHERE b.price > 9000;

EXPLAIN ANALYZE
SELECT
	f.flight_Id,
	b.price
FROM booking b
JOIN booking_flight bf
	ON b.booking_id = bf.booking_id
JOIN flights f
	ON bf.flight_id = f.flight_id
WHERE b.price > 9000;

CREATE UNIQUE INDEX idx_airline_code_unique
ON airline(airline_code);

SELECT
	airline_code,
	COUNT(*) AS duplicate_count
FROM airline
GROUP BY airline_code
HAVING COUNT(*) > 1;
